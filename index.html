Coloque o nome EXACTO do seu ficheiro .pdb no GitHub // (Atenção às maiúsculas/minúsculas e se tem "ll" ou "tt") let fileName = 'vitellogenin.pdb'; fetch(fileName) .then(response => { if (!response.ok) { throw new Error(`Ficheiro "${fileName}" não encontrado. Verifique o nome exato no GitHub.`); } return response.text(); }) .then(data => { // 2. Mudar a especificação de formato para "pdb" viewer.addModel(data, "pdb"); // 3.
